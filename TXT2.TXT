### 🚀 5 High-Value Logic Upgrades

#### 1. Upgrade `StateManager` to a Command Pattern (Delta-Based Undo)
**The Problem:** Your current `buildSnapshot()` performs a full deep-copy of the entire application state (walls, fixtures, openings, styles) on every action. For a floor plan with 100+ items, this will cause noticeable garbage collection (GC) spikes and UI jank.
**The High-Value Fix:** Instead of saving *states*, save *commands* (actions). Replaying 50 lightweight actions is exponentially faster than deep-cloning 50 massive states.

```javascript
// state-manager.js (High-Value Upgrade)
export class StateManager {
  constructor(appState, onRestore) {
    this.appState = appState;
    this.onRestore = onRestore;
    this.history = [];
    this.historyIndex = -1;
    this.maxHistory = 100; // Increased capacity due to smaller payload
  }

  // Execute an action and record it
  execute(command) {
    // 1. Apply the change to the live state
    command.execute(this.appState);
    
    // 2. If we undid previously, truncate the future
    if (this.historyIndex < this.history.length - 1) {
      this.history = this.history.slice(0, this.historyIndex + 1);
    }
    
    // 3. Push the command (not the whole state)
    this.history.push(command);
    this.historyIndex++;
    
    if (this.history.length > this.maxHistory) {
      this.history.shift();
      this.historyIndex--;
    }
    
    this.onRestore(this.appState);
  }

  undo() {
    if (this.historyIndex >= 0) {
      const command = this.history[this.historyIndex];
      command.undo(this.appState); // Command knows how to reverse itself
      this.historyIndex--;
      this.onRestore(this.appState);
    }
  }
}

// Example Command:
class AddWallCommand {
  constructor(wallData) { this.wallData = wallData; }
  execute(state) { state.walls.push(this.wallData); }
  undo(state) { state.walls.pop(); } // Or filter by ID
}
```

export function generateStairMesh(totalHeight, availableRun, width) {
  const MAX_RISER = 0.196; // ~7.75 inches in meters
  const MIN_TREAD = 0.254; // ~10 inches in meters

  // Calculate optimal number of steps
  let numSteps = Math.ceil(totalHeight / MAX_RISER);
  let actualRiser = totalHeight / numSteps;
  let actualTread = availableRun / (numSteps - 1); // One less tread than risers

  const stairGroup = new THREE.Group();
  const stepGeo = new THREE.BoxGeometry(width, actualRiser, actualTread);
  const stepMat = new THREE.MeshStandardMaterial({ color: 0x8B4513, roughness: 0.8 });

  for (let i = 0; i < numSteps; i++) {
    const step = new THREE.Mesh(stepGeo, stepMat);
    step.position.set(0, (i * actualRiser) + (actualRiser / 2), -(i * actualTread));
    step.castShadow = true;
    step.receiveShadow = true;
    stairGroup.add(step);
  }

  // Add stringers (side supports) procedurally here...
  return { mesh: stairGroup, specs: { numSteps, actualRiser, actualTread } };

#### 3. Smart Room Detection (Wall Graph to Polygon)
**The Problem:** Your `FloorPlanProcessor` excellently extracts walls, but the resulting `rooms: []` array is empty. Users expect the software to automatically detect enclosed spaces for flooring and area calculation.
**The High-Value Fix:** Implement a graph-traversal algorithm to find inner loops (enclosed polygons) formed by the extracted wall segments.

// floorPlanProcessor.js (Add this method)
detectRooms(walls) {
  const rooms = [];
  // 1. Build an adjacency graph of wall endpoints
  const graph = new Map();
  walls.forEach(wall => {
    const startKey = `${wall.start.x},${wall.start.y}`;
    const endKey = `${wall.end.x},${wall.end.y}`;
    if (!graph.has(startKey)) graph.set(startKey, []);
    if (!graph.has(endKey)) graph.set(endKey, []);
    graph.get(startKey).push({ point: wall.end, wallId: wall.id });
    graph.get(endKey).push({ point: wall.start, wallId: wall.id });
  });

  // 2. Traverse the graph to find closed loops (simplified concept)
  // In production, use a library like 'polybooljs' or a Half-Edge Data Structure
  // to perform boolean operations and extract inner faces.
  
  return rooms; // Array of { id, vertices: [[x,y], ...], area: number }
}

#### 4. 2D Spatial Hash Grid for Drag Collision & Snapping
**The Problem:** Dragging cabinets in 2D/3D currently relies on basic distance checks. In dense floor plans, this allows overlapping furniture or walls, breaking the illusion of a professional tool.

// scene-drag.js (High-Value Upgrade)
function checkCollision(draggedItem, allOtherItems) {
  const itemBox = new THREE.Box3().setFromObject(draggedItem.mesh);
  
  for (const other of allOtherItems) {
    if (other.id === draggedItem.id) continue;
    const otherBox = new THREE.Box3().setFromObject(other.mesh);
    
    // Expand otherBox slightly by wall thickness or minimum clearance
    otherBox.expandByScalar(0.1); 
    
    if (itemBox.intersectsBox(otherBox)) {
      return true; // Collision detected
    }
  }
  return false;
}

#### 5. Promote `InstancedMesh` for Performance at Scale
**The Problem:** Your `LabInstancing.js` shows you know about this, but the main `KitchenScene` or `FloorPlan3D` likely still creates individual `THREE.Mesh` objects for every wall segment, floor tile, and stair tread. This will choke the GPU at 500+ objects.
**The High-Value Fix:** Use `THREE.InstancedMesh` for repetitive geometry.

```javascript
// Example: Instanced Wall Segments
const wallCount = walls.length;
const geometry = new THREE.BoxGeometry(1, 2.7, 0.15); // Base geometry
const material = new THREE.MeshStandardMaterial({ color: 0xe0e0e0 });
const instancedWalls = new THREE.InstancedMesh(geometry, material, wallCount);

const dummy = new THREE.Object3D();
walls.forEach((wall, i) => {
  dummy.position.set(wall.position.x, wall.position.y, wall.position.z);
  dummy.rotation.set(0, wall.angle, 0);
  dummy.scale.set(wall.length, 1, 1);
  dummy.updateMatrix();
  instancedWalls.setMatrixAt(i, dummy.matrix);
});

instancedWalls.castShadow = true;
instancedWalls.receiveShadow = true;
scene.add(instancedWalls); // 1 draw call instead of N draw calls
```
